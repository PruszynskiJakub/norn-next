```yaml
description: Open or update the PR, then wait for its CI.
start: open-pr
steps:
  open-pr:
    type: git
    action: open-pr
    outcomes: { done: ci }
  ci:
    type: check
    run: ./scripts/wait-for-ci.sh
    timeout: 30m
    exits: { 3: stale }
    outcomes:
      pass:  { exit: green }
      fail:  { exit: red }
      stale: { exit: behind }
---
name: docs
start: implement
counters: { test_loop: { max: 3 }, ci_loop: { max: 3 }, stale_loop: { max: 3 } }
steps:
  implement:
    type: agent
    skill: norn-implement
    outcomes: { done: test }
  test:
    type: check
    run: npm ci && npm test
    timeout: 15m
    outcomes: { pass: update, fail: { to: implement, counter: test_loop } }
  update:
    type: git
    action: update
    outcomes: { done: test-merge, conflict: { to: implement, counter: test_loop } }
  test-merge:
    like: test
    outcomes: { pass: describe, fail: { to: implement, counter: test_loop } }
  describe:
    type: agent
    skill: norn-pr
    outcomes: { done: ship }
  ship:
    use: myship
    outcomes:
      green:  ready
      red:    { to: implement, counter: ci_loop }
      behind: { to: update, counter: stale_loop }
  ready:
    type: gate
    outcomes: { approve: $done, reject: { to: implement, counter: test_loop } }
```
